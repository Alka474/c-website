Enter student details like roll name, fee, DOB and print all the detail by ascending order of roll number ?

#include 
#include 

struct Student {
    int roll;
    char name[50];
    float fee;
    char dob[20]; // DD/MM/YYYY
};

void main() {
    struct Student s[50], temp;
    int i, j, n;
    
    clrscr();
    
    printf("How many students? ");
    scanf("%d", &n);
    
    // Input
    for(i = 0; i < n; i++) {
        printf("\nStudent %d:\n", i+1);
        printf("Enter Roll No: ");
        scanf("%d", &s[i].roll);
        
        printf("Enter Name: ");
        scanf("%s", s[i].name);
        
        printf("Enter Fee: ");
        scanf("%f", &s[i].fee);
        
        printf("Enter DOB (DD/MM/YYYY): ");
        scanf("%s", s[i].dob);
    }
    
    // Ascending Order Sort by Roll No - Bubble Sort
    for(i = 0; i < n-1; i++) {
        for(j = 0; j < n-i-1; j++) {
            if(s[j].roll > s[j+1].roll) {
                temp = s[j];
                s[j] = s[j+1];
                s[j+1] = temp;
            }
        }
    }
    
    // Print in Tabular Form
    printf("\n\n--- STUDENT DETAILS (Sorted by Roll No) ---\n");
    printf("%-10s %-20s %-15s %-15s\n", "Roll No", "Name", "Fee", "DOB");
    printf("----------------------------------------------------------------\n");
    
    for(i = 0; i < n; i++) {
        printf("%-10d %-20s %-15.2f %-15s\n", s[i].roll, s[i].name, s[i].fee, s[i].dob);
    }
    
    getch();
}

Factorial, Reverse, Palindrome, Prime, Armstrong, Fibonacci using Local and Global variable

#include 

int num;

void factorial() {
    int fact=1, i; // Local variable
    for(i=1; i<=num; i++) fact *= i;
    printf("Factorial = %d\n", fact);
}
void reverse() {
    int rev=0, temp=num, rem; // Local
    while(temp>0) {
        rem = temp%10;
        rev = rev*10 + rem;
        temp/=10;
    }
    printf("Reverse = %d\n", rev);
}
void palindrome() {
    int rev=0, temp=num, rem;
    while(temp>0) {
        rem = temp%10;
        rev = rev*10 + rem;
        temp/=10;
    }
    if(rev == num) printf("Palindrome\n");
    else printf("Not Palindrome\n");
}
void prime() {
    int i, count=0;
    for(i=1; i<=num; i++) if(num%i==0) count++;
    if(count==2) printf("Prime\n");
    else printf("Not Prime\n");
}
void armstrong() {
    int temp=num, rem, sum=0;
    while(temp>0) {
        rem = temp%10;
        sum += rem*rem*rem;
        temp/=10;
    }

 Swap numbers using Call by Address / Call by Function (Same hi hai)


#include 
void swap(int *a, int *b) {
    int temp;
    temp = *a;
    *a = *b;
    *b = temp;
}
int main() {
    int x, y;
    printf("Enter two numbers: ");
    scanf("%d %d", &x, &y);
    printf("Before Swap: x=%d y=%d\n", x, y);
    swap(&x, &y);
    printf("After Swap: x=%d y=%d\n", x, y);
    return 0;
}
